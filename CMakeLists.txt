cmake_minimum_required(VERSION 3.20)
project(NeoXR VERSION 0.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(NeoXR-CoreTest tests/core_test.cpp)
target_include_directories(NeoXR-CoreTest PRIVATE src)
add_test(NAME core COMMAND NeoXR-CoreTest)
add_executable(NeoXR-ModelTest tests/model_test.cpp)
target_include_directories(NeoXR-ModelTest PRIVATE src)
add_test(NAME model COMMAND NeoXR-ModelTest "${CMAKE_CURRENT_SOURCE_DIR}/assets/wheel.neo")
if(NOT WIN32)
  message(STATUS "Only portable core tests are available on this platform")
  return()
endif()
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Visual Studio 2022 or newer with the Windows SDK, x64")
endif()
add_executable(NeoXR-LayerTest tests/layer_test.cpp)
target_include_directories(NeoXR-LayerTest PRIVATE src)
target_compile_definitions(NeoXR-LayerTest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(NeoXR-LayerTest PRIVATE /W4 /EHsc)
target_link_libraries(NeoXR-LayerTest PRIVATE dinput8 dxguid user32)
add_test(NAME layer COMMAND NeoXR-LayerTest)
set(OPENXR_INCLUDE_DIR "" CACHE PATH "Folder containing openxr/openxr.h; leave empty to download official 1.0.34 headers")
if(NOT OPENXR_INCLUDE_DIR)
  include(FetchContent)
  if(POLICY CMP0169)
    cmake_policy(SET CMP0169 OLD)
  endif()
  FetchContent_Declare(openxr_headers
    URL https://github.com/KhronosGroup/OpenXR-SDK/archive/refs/tags/release-1.0.34.zip)
  FetchContent_GetProperties(openxr_headers)
  if(NOT openxr_headers_POPULATED)
    FetchContent_Populate(openxr_headers)
  endif()
  set(OPENXR_INCLUDE_DIR "${openxr_headers_SOURCE_DIR}/include")
endif()
add_library(NeoXR SHARED src/layer.cpp)
target_include_directories(NeoXR PRIVATE "${OPENXR_INCLUDE_DIR}" src)
target_compile_definitions(NeoXR PRIVATE UNICODE _UNICODE NEOXR_VERSION="${PROJECT_VERSION}")
target_compile_options(NeoXR PRIVATE /W4 /EHsc)
target_link_libraries(NeoXR PRIVATE d3d11 d3dcompiler dinput8 dxguid user32 windowscodecs ole32 winmm)
add_executable(NeoXR-Input src/inspect.cpp)
target_compile_definitions(NeoXR-Input PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(NeoXR-Input PRIVATE /W4 /EHsc)
target_link_libraries(NeoXR-Input PRIVATE dinput8 dxguid user32)
add_library(NeoXR-InputBridge SHARED src/input_bridge.cpp)
target_include_directories(NeoXR-InputBridge PRIVATE src)
target_compile_definitions(NeoXR-InputBridge PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(NeoXR-InputBridge PRIVATE /W4 /EHsc)
target_link_libraries(NeoXR-InputBridge PRIVATE dinput8 dxguid user32)
set(CMAKE_INSTALL_PREFIX "${CMAKE_BINARY_DIR}/package" CACHE PATH "Install directory" FORCE)
install(TARGETS NeoXR NeoXR-Input NeoXR-InputBridge RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES NeoXR.json NeoXR.ini Register.ps1 Unregister.ps1 README.md DESTINATION .)

install(DIRECTORY assets DESTINATION .)
