cmake_minimum_required(VERSION 3.20)
project(NeoXR VERSION 0.3.0 LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_FBX_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_OBJ_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_GLTF_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_NO_EXPORT ON CACHE BOOL "" FORCE)
set(ASSIMP_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(assimp URL https://github.com/assimp/assimp/archive/refs/tags/v5.4.3.zip)
FetchContent_MakeAvailable(assimp)
add_library(NeoXR-Model STATIC src/model.cpp)
target_include_directories(NeoXR-Model PUBLIC src)
target_link_libraries(NeoXR-Model PUBLIC assimp)
add_executable(NeoXR-CoreTest tests/core_test.cpp)
target_include_directories(NeoXR-CoreTest PRIVATE src)
add_test(NAME core COMMAND NeoXR-CoreTest)
add_executable(NeoXR-ModelTest tests/model_test.cpp)
target_include_directories(NeoXR-ModelTest PRIVATE src)
target_link_libraries(NeoXR-ModelTest PRIVATE NeoXR-Model)
add_test(NAME model COMMAND NeoXR-ModelTest "${CMAKE_CURRENT_SOURCE_DIR}/assets/neo-x.gltf" "${assimp_SOURCE_DIR}/test/models/FBX/box.fbx")
if(NOT WIN32)
  message(STATUS "Only portable core tests are available on this platform")
  return()
endif()
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Visual Studio 2022 or newer with the Windows SDK, x64")
endif()
add_executable(NeoXR-RenderTest tests/render_test.cpp)
target_compile_definitions(NeoXR-RenderTest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(NeoXR-RenderTest PRIVATE NeoXR-Model d3d11 d3dcompiler windowscodecs ole32 winmm)
add_test(NAME render COMMAND NeoXR-RenderTest "${CMAKE_CURRENT_SOURCE_DIR}/assets/neo-x.gltf" "${CMAKE_BINARY_DIR}/wheel-preview.ppm")
add_executable(NeoXR-LayerTest tests/layer_test.cpp)
target_include_directories(NeoXR-LayerTest PRIVATE src)
target_compile_definitions(NeoXR-LayerTest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(NeoXR-LayerTest PRIVATE /W4 /EHsc)
target_link_libraries(NeoXR-LayerTest PRIVATE dinput8 dxguid user32)
add_test(NAME layer COMMAND NeoXR-LayerTest)
set(OPENXR_INCLUDE_DIR "" CACHE PATH "Folder containing openxr/openxr.h; leave empty to download official 1.0.34 headers")
if(NOT OPENXR_INCLUDE_DIR)
  include(FetchContent)
  if(POLICY CMP0169)
    cmake_policy(SET CMP0169 OLD)
  endif()
  FetchContent_Declare(openxr_headers
    URL https://github.com/KhronosGroup/OpenXR-SDK/archive/refs/tags/release-1.0.34.zip)
  FetchContent_GetProperties(openxr_headers)
  if(NOT openxr_headers_POPULATED)
    FetchContent_Populate(openxr_headers)
  endif()
  set(OPENXR_INCLUDE_DIR "${openxr_headers_SOURCE_DIR}/include")
endif()
add_library(NeoXR SHARED src/layer.cpp)
target_include_directories(NeoXR PRIVATE "${OPENXR_INCLUDE_DIR}" src)
target_compile_definitions(NeoXR PRIVATE UNICODE _UNICODE NEOXR_VERSION="${PROJECT_VERSION}")
target_compile_options(NeoXR PRIVATE /W4 /EHsc)
target_link_libraries(NeoXR PRIVATE d3d11 d3dcompiler dinput8 dxguid user32 windowscodecs ole32 winmm)
target_link_libraries(NeoXR PRIVATE NeoXR-Model)
add_executable(NeoXR-Input src/inspect.cpp)
target_compile_definitions(NeoXR-Input PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(NeoXR-Input PRIVATE /W4 /EHsc)
target_link_libraries(NeoXR-Input PRIVATE dinput8 dxguid user32)
add_library(NeoXR-InputBridge SHARED src/input_bridge.cpp src/preview_bridge.cpp)
target_include_directories(NeoXR-InputBridge PRIVATE src)
target_compile_definitions(NeoXR-InputBridge PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(NeoXR-InputBridge PRIVATE /W4 /EHsc)
target_link_libraries(NeoXR-InputBridge PRIVATE d3d11 d3dcompiler dinput8 dxguid user32 windowscodecs ole32 winmm)
target_link_libraries(NeoXR-InputBridge PRIVATE NeoXR-Model)
set(CMAKE_INSTALL_PREFIX "${CMAKE_BINARY_DIR}/package" CACHE PATH "Install directory" FORCE)
install(TARGETS NeoXR NeoXR-Input NeoXR-InputBridge RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES NeoXR.json NeoXR.ini Register.ps1 Unregister.ps1 README.md THIRD-PARTY-NOTICES.txt Install-Desktop-Shortcut.ps1 DESTINATION .)

install(DIRECTORY assets DESTINATION .)
